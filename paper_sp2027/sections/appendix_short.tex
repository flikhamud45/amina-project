\section{Spot-Checking: Statements and Proofs}
\label{app:attack}

\para{Setting.} We use Definition~\ref{def:spot}, whose oracle defines the \emph{ideal-oracle
model}; $\tau(\theta)$ is the honest trace, $\mathcal A$ the set of admissible instances and
$\Sigma(\theta)$ the commitment oracle. The adversary $A$
sees $\theta$ and outputs $(u,v)$ with $v\neq h_u$ (conditioned on not aborting); $J$ is the edited
site. The forgery $\tau'$ is $\tau(\theta)$ with $h_u$ set to $v$ and later layers recomputed
with the true weights. It is \emph{explainable} if $\tau(\theta')=\tau'$ for some
$\theta'\in\mathcal A$ with $\theta'_{-u}=\theta_{-u}$.

\begin{definition}[location-oblivious rows, (H)]
\label{def:H}
(Prior, $A$) satisfies (H) if for every view $p$ there is an oracle $R_p$, possibly sampled lazily,
such that for every $u$ with $\Pr[J=u\mid P=p]>0$ and every verifier, the run of $V$ under
$\Sigma(\theta)\mid(P=p,J=u)$, up to and including the \emph{choice} of its first query that
touches $u$, has the same law as under $R_p$.
\end{definition}

\begin{theorem}[full form of Theorem~\ref{thm:guess}]
\label{thm:guess-full}
Let $G_t(J\mid P)=\mathbb E_p\max\{\sum_ux_u\Pr[J=u\mid P=p]:x\in[0,1]^N,\sum_ux_u\le t\}$. Assume
(H) and that every forgery of $A$ is explainable. In the ideal-oracle model, every
$V\in\mathrm{Spot}_d(q)$ rejects with probability at most $G_{qd}(J\mid P)\le
qd\cdot2^{-\tilde H_\infty(J\mid P)}$ (average min-entropy~\cite{dodis2008fuzzy}), plus
$2Q2^{-\kappa}+Q^22^{-\ell_H}$ under the second condition of Remark~\ref{rem:merkle}.
\end{theorem}
\begin{proof}[Proof sketch]
Completing the proof idea of Theorem~\ref{thm:guess}: whether $V$ touches $u$ is fixed by its
run up to the choice of that query, so by (H) $x_u=\Pr_{R_p}[V\text{ touches }u]$ for every $u$.
The min-entropy bound follows termwise. For $d=1$, opening the $q$ most likely sites attains $G_q$,
as binding makes row $J$ contradict $v$. The full proof is in the extended
version~\cite{fullversion}.
\end{proof}

\begin{lemma}[independent rows give (H)]
\label{lem:product}
If the prior on $\theta$ given $Z$ is a product $\bigotimes_j\mu_j$ and the law of $A(\theta)$
depends only on $h(\theta)$, then (H) holds with $R_p=\bigotimes_j\mu_j(\cdot\mid f_j(\theta_j)=p_j)$.
\end{lemma}

\begin{corollary}[the posterior on the edited site]
\label{cor:surprise}
Under Lemma~\ref{lem:product}, let $\pi_u$ be the law of $f_u(\theta_u)$, $\theta_u\sim\mu_u$, and
$\beta_u(p)=\Pr_{h_u\sim\pi_u}[A(h_u,p_{-u})=(u,p_u)]$. Then $\Pr[J=u\mid P=p]\propto
\beta_u(p)/\pi_u(p_u)$, a point mass on $u$ if $\pi_u(p_u)=0$.
\end{corollary}

\noindent On $\{P=p,J=u\}$ the law of $\theta$ factorises over $\theta_{-u}$ and $\theta_u$, so
$\Pr[J=u,P=p]=(\prod_j\pi_j(p_j))\beta_u(p)/\pi_u(p_u)$. The full proofs are in the extended
version~\cite{fullversion}.

\begin{proposition}[the hard instance; Theorem~\ref{thm:dich}(a)]
\label{prop:worst}
Let $N\ge2$ and $\sigma$ be a uniform permutation of $[N]$. Hidden neuron $i$ has the row
$e_{\sigma(i)}$ and bias $0$, $g=\mathrm{ReLU}(\sum_ih_i-(N-1))$, and the logits are $(1/2,g)$;
natural inputs are uniform over $\mathbf 1$ and the $N$ vectors with one zero. With a commitment
that hides unopened rows, every $V\in\mathrm{Spot}_d(q)$ that knows the later weights, the natural
ranges and each hidden neuron's natural distribution has $\mathbb E_\sigma\Pr[V\text{
accepts}]\ge1-qd/N$. For an unsalted Merkle tree ($\ell_H$-bit hashes, $Q$ hash evaluations by
$V$), give each hidden row independent uniform weights on $c\ge\kappa/b$ dead input coordinates, zero on every
natural input ($b$ bits per weight); the bound becomes $1-qd/N-2Q2^{-\kappa}-Q^22^{-\ell_H}$.
\end{proposition}
\begin{proof}[Proof sketch]
The flip is as in the proof idea of Theorem~\ref{thm:dich}; row $e_j$, $j\neq\zeta$, explains
it. (H) holds with the $R_p$ that maps each newly touched row to a uniform unused element
of $[N]\setminus\{\zeta\}$. The view, including each hidden neuron's marginal, does not depend on
$\sigma$, so $J$ is uniform given it and Theorem~\ref{thm:guess-full} bounds detection by $qd/N$.
In the Merkle case the dead-input weights give each leaf preimage min-entropy $\kappa$
(Remark~\ref{rem:merkle}). The full proof is in the extended version~\cite{fullversion}.
\end{proof}

\begin{proposition}[value-blind samplers]
\label{prop:blind}
Let $V$ reject only on a failed local check and choose rows independently of the claimed values.
On a fixed network, let disjoint site sets $B_1,\dots,B_m$ carry forgeries $\tau'_j$ inconsistent
exactly on $B_j$. Then $\min_j\Pr[V\text{ rejects }\tau'_j]\le q/m$.
\end{proposition}

\noindent The opened rows are true, so until its first opening in $B_j$ the run on
$\tau'_j$ equals one reference run, whose at most $q$ expected openings meet the disjoint $B_j$ at
most $q$ times in total.

\begin{proposition}[mixing suffices]
\label{prop:amp}
Let a layer compute $Z=A[X;\mathbf 1]$, $A\in\F^{N\times k}$, without wrap-around, and let the
committed matrix hold the Reed--Solomon encodings, of length $n$, of the rows of $A$ (honest $C_M$;
after a setup proof of radius $e$, replace $k-1$ by $k-1+e$). A verifier that draws $r$ vectors
$\chi$ after $Z$, checks $\chi^\T Z=u[X;\mathbf 1]$ for the received folded rows $u$, and then
checks $\chi^\T\hat A_c=\enc(u)_c$ on $t$ distinct uniform columns, accepts a wrong $Z$ with
probability at most $p^{-r}+((k-1)/n)^t$ plus that of a hash collision. This is at most
$2^{1-\beta}$ for $r=\lceil\beta/\log_2p\rceil$ and $t=\lceil\beta/\log_2(n/(k-1))\rceil$, and every
query has support $[N]$.
\end{proposition}

\noindent The proof is that of Theorem~\ref{thm:sound} for one matrix (Appendix~\ref{app:sound}).

\begin{proof}[Proof of Theorem~\ref{thm:dich}]
(a)~By Proposition~\ref{prop:worst}, error $\varepsilon$ requires $1-qd/N\le\varepsilon$; a path
opens one row of the layer, so $s$ paths are $s$ queries with $d=1$. (b)~Set
$\beta=\lambda+\log_2(2L)$ in Proposition~\ref{prop:amp} with a union bound over the $L$ layers.
\end{proof}

\begin{remark}[commitments]
\label{rem:merkle}
The lower bounds need a commitment that \emph{hides} unopened rows. Unsalted Merkle paths over
low-entropy rows leak: on Proposition~\ref{prop:worst} without dead inputs, inverting sibling
subtrees up to height $\eta$ detects the edit with probability $2^{\eta+1}/N$ instead of $1/N$.
Theorem~\ref{thm:guess-full} holds for salted leaves $H(s_i\Vert\theta_i)$, and in the
random-oracle model for an unsalted tree if every leaf preimage, under the true rows and the
explanation, has min-entropy at least $\kappa$ given the view and the other leaves. Trained weights
have high-entropy rows, so this concerns the proof, not the attack.
\end{remark}

\begin{remark}[correlated rows]
\label{rem:pointer}
Let row $0$ store the index $z$ of the critical site in weights on dead inputs, and sites
$1,\dots,m$ carry bits whose only zero is at $z$, with their AND as output. The forged trace is the
same for every $z$, so $G_2=2/m$, yet an adaptive verifier opens row $0$, reads $z$ and opens row
$z$.
\end{remark}

\begin{remark}[scope]
\label{rem:scope}
(i)~Imperfect completeness $c$ adds $1-c$. (ii)~For edits explainable by the rows of the
inconsistent set $B$ alone, replace ``touch $J$'' by ``touch $B$''. (iii)~The bound is per query,
for a stateless verifier; caching opened rows amounts to downloading the layer after about $N/q$
queries. (iv)~Proposition~\ref{prop:worst} embeds in every ReLU MLP with input dimension and hidden
width at least $N$ and one further hidden layer. In a transformer the LayerNorm before the MLP
blocks the 0/1 patterns; we do not claim the embedding. (v)~Joint statistics are not covered.
\end{remark}

\section{The Defence: Semantics, Proofs and Parameters}
\label{app:defence}

\subsection{Integer Semantics}
\label{app:semantics}

A requantisation computes
$\mathrm{clamp}(\lfloor(mz+2^{29})/2^{30}\rfloor,-127,127)$; a residual addition adds
$\lfloor(mz+2^{29})/2^{30}\rfloor$ and clamps to $[1-2^{22},2^{22}-1]$. LayerNorm subtracts the
rounded integer mean, RMSNorm does not; both output
$\mathrm{clamp}(\lfloor(g_ix_i+2^{15}\sigma)/(2^{16}\sigma)\rfloor,-127,127)$ with public integer
gains $g_i$ and the exact $\sigma=\max(1,\lfloor\sqrt{\lfloor\sum_ix_i^2/d\rfloor}\rfloor)$.
Activations are 256-entry int8 tables. Rotary embeddings rotate pairs by
$\mathrm{round}(2^{14}\cos t\theta_j)$ and $\mathrm{round}(2^{14}\sin t\theta_j)$, then round and
clamp. Attention maps the score gaps $g_j=\max_{j'}s_{j'}-s_j$, $s=qk^\T$, to
$i_j=\mathrm{clamp}(\lfloor(m_sg_j+2^{29})/2^{30}\rfloor,0,255)$ and
$e_j=\mathrm{round}(2^{15}e^{-i_j/16})$ ($0$ if masked); with $E=\sum_je_j$,
$P_j=\lfloor(510e_j+E)/(2E)\rfloor\in[0,255]$, and $PV$ is requantised. Arithmetic is exact: int8
products sum in 32-bit blocks of $2^{16}$, and attention runs in float32 below $2^{24}$ (scores at
most $2^{21}$, partial sums of $PV$ below $(255+T/2)\cdot128$ for $T<2^{15}$). Full details are in
the extended version~\cite{fullversion}.

\subsection{Proof of Theorem~\ref{thm:sound}}
\label{app:sound}

Let $D_\ell=Z_\ell-A_\ell[\tilde X_\ell;\mathbf 1]$ over $\F$, with $\tilde X_\ell$ the verifier's
derived input. If $y\neq M(x)$, the first operation $\ell$ with a wrong claim has the true
input, and the range checks give $D_\ell\neq0$ (Section~\ref{sec:intmodel}). Barring a collision it
is not a lookup, so it belongs to some $j\in\mathcal J$.

\para{Row layout.} The sketch after Theorem~\ref{thm:sound} applies: barring a collision the opened
columns are honest and drawn after $u_\ell$.

\para{Transposed layout.} The verifier checks $\enc(Z\chi'_i)_c=w_i^\T\hat A'_c$ with
$w_i=[\tilde X;\mathbf 1]\chi'_i$; honest claims pass since $w_i^\T\hat A'=\enc(Aw_i)$. If
$D\chi'_i=0$ for all $r$ columns, a nonzero row of $D$ is orthogonal to $r$ uniform vectors
(probability $p^{-r}$); otherwise the column term applies.

\para{Modes $\sK$ and $\Kpre$.} In $\sK$ only the first case arises. In $\Kpre$, against a hybrid
verifier that reports the outcomes predicted from the prover's messages, the
messages are independent of the secret vectors. A passing check with $D_\ell\neq0$ thus occurs
among $Q$ queries with probability at most $Q\sum_\ell p^{-r}$, and the real run equals the hybrid
until then.

\para{Round-by-round soundness.} A partial transcript is \emph{doomed} if (i)~after the claims,
$y\neq M(x)$; (ii)~after the vectors, some $\chi_j^\T D_j\neq0$ or $D_j\chi'_j\neq0$; (iii)~after
the folds, a Freivalds check fails, some $u_j\neq\chi_j^\T A_j$, or some $D_j\chi'_j\neq0$. Each
challenge leaves a doomed state with probability at most $p^{-r}$ or one column term, which gives
the bounds of Section~\ref{sec:fs}. The full proof is in the extended version~\cite{fullversion}.

\subsection{Proof Idea for Theorem~\ref{thm:untrusted}}
\label{app:untrusted}

\para{Extraction.} In the random-oracle model, the queries made before $C_M$ is output fix the
leaves under every root, barring a collision ($O(Q^22^{-256})$~\cite{bcs16}). A column that cannot
be extracted counts as corrupted.

\para{Setup.} If $E_j$ is not $e_j$-close, correlated agreement~\cite[Thm.~1.7]{bciks20}
\todo{FOCS numbering} makes $\psi^\T E_j$ $e_j$-far from the code over $\mathbb K_s$ for uniform $\psi\in\mathbb K_s^{R_j}$ except
with probability $n_j/|\mathbb K_s|$, and by Fact~\ref{fact:ext} the $s$ base-field combinations
form such a $\psi$. The checks at a column pass exactly where the sent codeword agrees with
$\psi^\T E_j$: on at most $n_j-e_j-1$ columns, fixed before the $t^0_j$ columns are drawn.
Fiat--Shamir multiplies the error by $Q$.

\para{Online.} Each $E_j$ differs from $\enc(A^*_j)$ on a set $B_j$ of at most $e_j$ columns fixed
by $C_M$. The first claim that differs from $M^*$ has $|z|<2^{29}<p/2$, and the value of $M^*$
lies in $(-p/2,p/2)$, so $D_\ell\neq0$ without a bound on $A^*$. The column term with $k_j+e_j$
follows as in Section~\ref{sec:setup-proof}, for every $w_i$ in the transposed layout, and the round-by-round argument
carries over. The full proof is in the extended version~\cite{fullversion}.

\section{Soundness of V1}
\label{app:v1}

\para{Notation and checks.} $S=\F+\xi\F$ has $p^2$ elements. Entry $(i,j)$ of a cut operation
$\ell$ has sent value $s_{ij}$, window start $L_{ij}$ and width $W_{ij}$ ($\Vlo_\ell(s_{ij})$ and
$W_\ell(s_{ij})$, or $z$ and $1$ for an exception $z$), residue
$\delta_{ij}=(A_\ell\hat X_\ell)_{ij}-L_{ij}\in\mathbb Z$ and leaf $v_{ij}=\delta_{ij}+\xi W_{ij}$.
The verifier checks (D1)~ranges ($|a|\le127$, residual values in
$[\Vshift_\ell(1-2^{29}),\Vshift_\ell(2^{29}-1)]$, clear claims below $2^{29}$); (D2)--(D3)~the
recomputation and the token rule; (D4)~that exceptions lie in cut requantisations,
increase strictly and have $|z|<2^{29}$ and $\Vrq_\ell(z)=a_{ij}$; (D5)~$\mu_\tau<p$ and
$\sum_\tau\mu_\tau=\Nleaf$ in $\mathbb Z$; (G)~the GKR rounds and $\mathsf d_0^\gamma\neq0$;
(F1)--(F2)~Freivalds' checks for clear operations and, in $\Kpre$, $\chi_s^\T y_\ell=u_s\bar x_\ell$;
(F3)~the final values~\eqref{eq:v1-final}; (F4)~$\sum_\gamma\mathsf n_0^\gamma/\mathsf
d_0^\gamma=\sum_\tau\mu_\tau/(\alpha-\tau)$; and (C)~the column check on the eight base-field
planes of each fold.

\begin{lemma}[windows]
\label{lem:v1-window}
For $m_\ell>0$, $\Vshift_\ell(z)=v$ if and only if $\Vlo_\ell(v)\le z<\Vlo_\ell(v+1)$, and
$W_\ell(v)\in\{\lfloor2^{30}/m_\ell\rfloor,\lceil2^{30}/m_\ell\rceil\}$. If (D1) and (D4) pass and
every residue of a cut operation lies in $[0,W_{ij})$, then $s_\ell$ is
$\Vrq_\ell(A_\ell\hat X_\ell)$ or, for a residual operation, $\Vshift_\ell(A_\ell\hat X_\ell)$.
\end{lemma}

\begin{lemma}[no wrap-around]
\label{lem:v1-nowrap}
If a cut operation has the honest input, (D1) and (D4) pass and $\delta_{ij}\notin[0,W_{ij})$, then
$v_{ij}\notin\Vtab$.
\end{lemma}

\begin{lemma}[fractional-sum GKR]
\label{lem:v1-gkr}
Fix $\alpha$ and the true leaves of instance $\gamma$. If the root computed from the prover's first
message is wrong, then, except with probability $\varepsilon_{\mathrm{GKR}}(n_\gamma)$, a check (G)
fails or the final claims differ from the true multilinear extensions at $\rho$.
\end{lemma}

\begin{lemma}[final values]
\label{lem:v1-final}
The true leaf layer of instance $\gamma$ satisfies~\eqref{eq:v1-final} with
$I(\rho)=I_{R_\gamma}(r_{\mathrm{row}})I_T(r_{\mathrm{col}})$, where $I_N(r)=\sum_{y<N}\Veq(r,y)$.
\end{lemma}

\para{Proof ideas.} For Lemma~\ref{lem:v1-logup}, unique partial fractions give leaf counts
$c_v\equiv0$ off the table and $c_\tau\equiv\mu_\tau\pmod p$ on it; as $0\le\mu_\tau<p$,
$c_\tau\ge\mu_\tau$, so $\sum_\tau\mu_\tau=\sum_vc_v$ forces every off-table count to zero.
Lemma~\ref{lem:v1-window} unfolds the floor. For Lemma~\ref{lem:v1-nowrap}, (P3), (P4), (D1) and
(D4) give $|\delta_{ij}|<2^{30}+2^{16}$, and $p-2^{30}-2^{16}=939{,}458{,}561>2^{16}\ge W_{ij}$, so
$\delta_{ij}\bmod p\notin[0,W_{ij})$. For Lemma~\ref{lem:v1-gkr}, the line through the sent
children, the combination with $\eta_k$ and the degree-3 sum-check~\cite{lfkn1992} of layer $k$ let
a false claim become true with probability at most $1/p^8$, $1/p^8$ and $3k/p^8$. For
Lemma~\ref{lem:v1-final}, $\Veq$ factorises, and the leaf in row $o_\ell+i$ and column $j$ has
weight $(\chi_\ell)_ie_j$. The full proofs are in the extended version~\cite{fullversion}.

\para{Proof of Theorem~\ref{thm:v1}.} Completing the sketch in Section~\ref{sec:v1}: if every
clear claim is correct and every residue lies in its window, Lemma~\ref{lem:v1-window} and induction
give the honest logits, so $\ell^\ast$ exists. The off-table leaf of a cut $\ell^\ast$
(Lemma~\ref{lem:v1-nowrap}) is fixed with $\mu$ before $\alpha$. By (D5) and
Lemma~\ref{lem:v1-logup}, $\sum_i1/(X-v_i)-\sum_\tau\mu_\tau/(X-\tau)$ is nonzero with at most
$\Nleaf+|\Vtab|$ poles, all in $S$, so it vanishes at $\alpha\in\Fext\setminus S$ with probability
at most $(\Nleaf+|\Vtab|)/(p^8-p^2)$. Otherwise (F4), Lemma~\ref{lem:v1-gkr} and
Lemma~\ref{lem:v1-final} make (F3) fail unless a fold is wrong in some base-field plane, which the
column check catches except with the column term. In $\Kpre$ the secret vectors are handled as in
Appendix~\ref{app:sound}. Under Fiat--Shamir the new rounds err with probability at most $1/p^8$,
$3/p^8=2^{-245.7}$ and $2^{-215.3}$. The full proof is in the
extended version~\cite{fullversion}.
