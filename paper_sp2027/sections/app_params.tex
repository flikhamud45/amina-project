\subsection{Concrete Security for Llama-2-7B}
\label{app:params}

Table~\ref{tab:params} gives each term of Theorems~\ref{thm:sound}, \ref{thm:untrusted}
and~\ref{thm:v1}.

\begin{table}[t]
  \caption{Concrete security for Llama-2-7B at $\lambda=128$ (interactive / Fiat--Shamir), under the
  commitment plan of our runs (every codeword at least $2^{18}$ points)}
  \label{tab:params}
  \centering
  \footnotesize
  \setlength{\tabcolsep}{3pt}
  \begin{tabular}{@{}p{5.0cm}l@{}}
    \toprule
    Term and rule & Value \\
    \midrule
    \multicolumn{2}{@{}l}{\emph{Per query, v0 (Theorem~\ref{thm:sound}, Section~\ref{sec:fs})}} \\
    Field $\F$, $p=15\cdot2^{27}+1$ & $\log_2p\approx30.9$ \\
    Weight operations $n_{\mathrm{op}}\ge|\mathcal J|$ & 226 \\
    Target $\beta=\lambda+\log_2(2n_{\mathrm{op}})$, plus 64 under Fiat--Shamir & 136.8 / 200.8 \\
    Freivalds vectors $r=\lceil\beta/\log_2p\rceil$ & 5 / 7 \\
    Freivalds term $p^{-r}$ per matrix & $2^{-154.5}$ / $2^{-216.3}$ \\
    Columns $t_j$: least $t$ with $\prod_{i<t}\frac{k_j-1-i}{n_j-i}\le2^{-\beta}$ &
      23 / 34$^a$ \\
    Opened columns per query & 65.6 / 96.5\,MB \\
